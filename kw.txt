SELECT idObiekt, plik, nazwa
FROM obiekty
WHERE panstwo = 'Islandia';

kw2
SELECT plik, nazwa, nazwaCechy, wartoscCechy, opis, rodzaje.rodzaj
FROM obiekty
JOIN rodzaje
ON obiekty.idRodzaj = rodzaje.idRodzaj
WHERE obiekty.idObiekt = 46;

kw3
SELECT obiekty.nazwa
FROM obiekty
JOIN rodzaje
ON obiekty.idRodzaj = rodzaje.idRodzaj
WHERE obiekty.panstwo = 'Islandia'
AND rodzaje.idRodzaj = 10;

kw4
SELECT rodzaje.rodzaj, COUNT(*) AS 'Liczba obiektów'
FROM obiekty
JOIN rodzaje
ON obiekty.idRodzaj = rodzaje.idRodzaj
GROUP BY rodzaje.idRodzaj, rodzaje.rodzaj;
